import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

similarity = cosine_similarity(embeddings)

# argsort 后取最后五个，再反转为从高到低。

Task 13 — 点击 passage 后寻找 5 个最近语义邻居

本页在简化 embedding 上计算；真实作业应使用模型生成的完整向量。

function cosineSimilarity(a, b) {
  const dot = d3.sum(a.map((value, i) => value * b[i]));
  const lengthA = Math.sqrt(d3.sum(a.map(value => value * value)));
  const lengthB = Math.sqrt(d3.sum(b.map(value => value * value)));
  return dot / (lengthA * lengthB);
}

function nearestNeighbors(selected, data, count = 5) {
  return data
    .filter(d => d.passage_id !== selected.passage_id)
    .map(d => ({ ...d, similarity: cosineSimilarity(selected.embedding, d.embedding) }))
    .sort((a, b) => d3.descending(a.similarity, b.similarity))
    .slice(0, count);
}
