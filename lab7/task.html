// task.json 是由课程提供的天气 CSV 整理而来。
// JSON 中 date 仍是字符串，所以读取后必须转换成 Date。
const parseDate = d3.timeParse("%Y-%m-%d");
d3.json("task.json").then(rawData => {
  const data = rawData.map(d => ({ ...d, date: parseDate(d.date) }));
  // data 现在可以用于 scaleTime、extent 和 bisector。

Task 3：单城市折线图
// 先取 Tokyo，按日期升序排列；这就是最小的折线图数据。
const cityData = data.filter(d => d.city === "Tokyo")
  .sort((a, b) => d3.ascending(a.date, b.date));

const xScale = d3.scaleTime()
  .domain(d3.extent(cityData, d => d.date))
  .range([margin.left, width - margin.right]);
const yScale = d3.scaleLinear()
  .domain(d3.extent(cityData, d => d.temperature_c)).nice()
  // 屏幕 y 轴向下增加，所以 range 必须反过来。
  .range([height - margin.bottom, margin.top]);

const line = d3.line()
  .x(d => xScale(d.date))
  .y(d => yScale(d.temperature_c));
svg.append("path").datum(cityData).attr("d", line);
Task 4：比较多座城市
const selectedCities = ["Tokyo", "London", "New York"];
const filteredData = data.filter(d => selectedCities.includes(d.city));
const grouped = d3.groups(filteredData, d => d.city);
const colorScale = d3.scaleOrdinal()
  .domain(selectedCities).range(d3.schemeTableau10);

// grouped 的每个元素是 [城市名, 该城市的观测数组]。
svg.selectAll(".city-line").data(grouped, d => d[0]).join("path")
  .attr("class", "city-line").attr("fill", "none")
  .attr("stroke", d => colorScale(d[0]))
  .attr("d", d => line(d[1]));
Task 5：切换天气指标
// 下拉选单的 value 是资料栏位名称，例如 temperature_c。
d3.select("#metric").on("change", function () {
  updateChart(this.value);
});

function updateChart(metric) {
  yScale.domain(d3.extent(filteredData, d => d[metric])).nice();
  line.y(d => yScale(d[metric]));
  svg.selectAll(".city-line").transition().duration(600)
    .attr("d", d => line(d[1]));
}
Task 6：悬停明细（tooltip）
const bisectDate = d3.bisector(d => d.date).center;
function moved(event) {
  const [mouseX] = d3.pointer(event);       // 像素位置
  const date = xScale.invert(mouseX);       // 像素位置转回日期
  const index = bisectDate(cityData, date); // 找最接近的资料点
  const d = cityData[index];
  tooltip.style("opacity", 1).html(`${d.city}<br>${d.date}`);
}
Task 7：日期范围与 brush
// 日期输入框筛选资料后，重新设定两个比例尺的 domain。
const rangeData = cityData.filter(d =>
  d.date >= startDate && d.date <= endDate);
xScale.domain(d3.extent(rangeData, d => d.date));
yScale.domain(d3.extent(rangeData, d => d.temperature_c)).nice();

const brush = d3.brushX().extent([[margin.left, margin.top],
  [width - margin.right, height - margin.bottom]])
  .on("end", brushed);
function brushed(event) {
  if (!event.selection) return;
  const [x0, x1] = event.selection;
  // xScale.invert 把框选的左右像素边界转换成起止日期。
  applyRange(xScale.invert(x0), xScale.invert(x1));
}
Task 8–9：动画概念与单帧
// 折线图：时间映射到 x 位置；动画：资料时间映射到播放顺序。
let currentIndex = 0;
let timer = null;
const marker = svg.append("circle").attr("r", 7).attr("fill", "#dc2626");

function showFrame(index) {
  const d = cityData[index];
  marker.attr("cx", xScale(d.date)).attr("cy", yScale(d.temperature_c));
  dateLabel.text(d3.timeFormat("%Y-%m-%d")(d.date));
  d3.select("#time-slider").property("value", index);
}
Task 10：播放、暂停、重置与拖动
function play() {
  if (timer) return; // 避免重复建立多个计时器。
  timer = d3.interval(() => {
    showFrame(currentIndex);
    currentIndex += 1;
    if (currentIndex >= cityData.length) pause();
  }, 150);
}
function pause() { if (timer) { timer.stop(); timer = null; } }
function reset() { pause(); currentIndex = 0; showFrame(0); }
d3.select("#time-slider").on("input", function () {
  pause(); currentIndex = +this.value; showFrame(currentIndex);
});
